def clean(x):
 if isinstance(x,np.ndarray):return x.tolist()
 if isinstance(x,np.generic):return x.item()
 if isinstance(x,dict):return {k:clean(v) for k,v in x.items()}
 if isinstance(x,(tuple,list)):return [clean(v) for v in x]
 return x

def save(name,x): (P/name).write_text(json.dumps(clean(x),indent=2))
def table(name,rows):
 fields=list(dict.fromkeys(k for x in rows for k in x))
 with (P/name).open('w') as f:
  w=csv.DictWriter(f,fields);w.writeheader();w.writerows(rows)

def bound(g):
 n=len(g['marks']);D=(2*math.ceil(.015*g['R'])+1)**2;M=[len(s) for s in g['supports']]
 labeled=math.prod(max(0,m-i*D) for i,m in enumerate(M))
 return {'support_sizes':M,'excluded_pixel_upper':D,'N_labeled_lower':str(labeled),'N_visible_lower':str(labeled//math.factorial(n))}

def toy_sampler():
 supports=[np.array([[0.,.6],[0.,.8],[.4,0.]]),np.array([[0.,.6],[0.,.8],[.4,0.]])];weights=collections.Counter();labeled=0
 for a in supports[0]:
  for b in supports[1]:
   if np.linalg.norm(a-b)<.015:continue
   labeled+=1;weights[tuple(sorted([tuple(a),tuple(b)]))]+=1
 assert labeled==6 and len(weights)==3 and set(weights.values())=={2}
 return {'labeled_admissible':labeled,'visible':len(weights),'multiplicity':2,'proposal_probability':'1/9 each labeled pair; conditional probability1/6; visible weight2/6','separation_rejection_pass':True,'sampler_uniformity':'Independent uniform integer-index selection gives uniform product-support proposals. Reject pair distances<.015. Every admissible labeled configuration has the same proposal probability, so accepted draws have uniform labeled measure.'}

  if time.monotonic()>deadline:raise RuntimeError('variant time cap')
  a=alpha if name.startswith('threshold') else render(source,N,angle)

 colors={'E':'#208cff','J':'#e07800','H':'#b000cc'}
 for j,m in enumerate(g['marks']):ax.plot(m['yx'][1],m['yx'][0],'o',ms=5,color=colors[m['class']]);ax.text(m['yx'][1]+6,m['yx'][0],m['class']+str(j+1),fontsize=7,color=colors[m['class']])
 ax.set_title(sign+' — '+('stable' if passed else 'unstable')+f' ({n} landmarks)');ax.set_xlim(0,1024);ax.set_ylim(1024,0);ax.axis('off');fig.tight_layout();fig.savefig(P/f'figures/{sign}.png',dpi=110);plt.close(fig)
 audits.append(f'<section><h3>{sign}: {"stable" if passed else "unstable"}, {n} landmarks</h3><img src="figures/{sign}.png"><p>{"; ".join(failures) if failures else "All mandatory representation checks passed."}</p></section>')
 print(sign,'landmarks',n,'stable',passed,'seconds',round(time.monotonic()-start,1),flush=True)

  nulls.append({'sign':sign,'status':'PARTIAL','accepted':len(accepted),'proposals':proposals,'seed':2026100600+i});continue

  med=float(np.median(a));effect=float(v-med);critical=float(np.quantile(a,1-.05/90,method='higher'));power=float(np.mean(a+.15>=critical))

  results.append({'sign':sign,'endpoint':e,'observed_descriptive':v,'null_median':med,'effect':effect,'p':pv,'p_Bonferroni90':padj,'CP_upper95':upper,'corrected_CP_upper95':min(1,90*upper),'null_min':float(a.min()),'null_max':float(a.max()),'largest_null_atom':max(counts[ei].values())/9999,'shift_calibration_power':power,'status':status,'reason':'frozen corrected/effect/uncertainty/power rule','inference':'RUN'})
