#!/usr/bin/env python3
"""Verify preserved upstream blobs and all archive files against recorded hashes."""
from pathlib import Path
import hashlib,json
root=Path(__file__).resolve().parent
retrieval=json.loads((root/'evidence/retrieval.json').read_text())
for f in retrieval['files']:
    data=(root/'upstream'/f['repo']/f['path']).read_bytes()
    tree=retrieval['trees'][0 if f['repo']=='rulebench' else 1]
    recorded=next(e['sha'] for e in tree['tree'] if e['path']==f['path'])
    actual=hashlib.sha1(b'blob '+str(len(data)).encode()+b'\0'+data).hexdigest()
    assert actual==f['sha']==recorded,f['path']
manifest=root/'SHA256SUMS.json'
if manifest.exists():
    for name,expected in json.loads(manifest.read_text()).items():
        assert hashlib.sha256((root/name).read_bytes()).hexdigest()==expected,name
print('Verified 22 upstream files against pinned tree blobs, and archive manifest when present.')
print('Hash agreement identifies bytes; it does not prove any mathematical claim or author identity.')
