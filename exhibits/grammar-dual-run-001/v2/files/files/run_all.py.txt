#!/usr/bin/env python3
"""Reexecute both miniature experiments plus root reuse. --benchmark adds CPU diagnostics."""
from pathlib import Path
import json, subprocess, sys
HOME=Path(__file__).resolve().parent

def run(rel):
    print(f'RUN {rel}',flush=True)
    subprocess.run([sys.executable,str(HOME/rel)],cwd=str((HOME/rel).parent),check=True)

if __name__=='__main__':
    run('fixed-grammar/grammar001.py')
    first=json.loads((HOME/'fixed-grammar/results.json').read_text())
    original=json.loads((HOME/'fixed-grammar/baseline-recorded.json').read_text())
    assert first==original,'Fixed grammar no longer matches original!'
    run('root-first/root_grammar001.py')
    run('root-first/reuse_extension.py')
    if '--benchmark' in sys.argv:
        run('root-first/benchmark.py')
    print('PASS: original record reproduced; 31 root test cases and 128 append-only queries independently checked.')
