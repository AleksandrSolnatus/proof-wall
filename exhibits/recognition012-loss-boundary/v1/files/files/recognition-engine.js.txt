/* AREPO 012 — Recognition under transformation. Pure deterministic geometry.
   No physical identity, consciousness, or metaphysical claims. */
(function(root,factory){
 const M=factory();
 if(typeof module==='object' && module.exports) module.exports=M;
 root.ArepoExtremeRecognition=M;
})(typeof globalThis!=='undefined'?globalThis:this,function(){
 'use strict';
 const TAU=2*Math.PI;
 const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));
 const defaults=()=>({scale:1,bend:0,rotation:0,split:false,separation:0,collapse:0,ledger:true});
 function validate(raw={}){
  const p={...defaults(),...raw};
  for(const k of ['scale','bend','rotation','separation','collapse']) if(!Number.isFinite(p[k]))throw Error('Non-finite '+k);
  p.scale=clamp(p.scale,.1,3);p.bend=clamp(p.bend,0,1.4);p.rotation=clamp(p.rotation,-180,180);
  p.separation=clamp(p.separation,0,150);p.collapse=clamp(p.collapse,0,1);
  p.split=!!p.split;p.ledger=!!p.ledger;return p;
 }
 // Undistorted radius=1. For 0<=bend<1, radial distance remains positive.
 function radial(t,a){return 1+a*Math.cos(3*t);}
 function sourcePoint(t){return [Math.cos(t),Math.sin(t)];}
 function mappedPoint(t,raw={},halfSign=0){
  const p=validate(raw),a=p.rotation*Math.PI/180;
  const r=p.scale*(1-p.collapse)*radial(t,p.bend);
  const x=r*Math.cos(t),y=r*Math.sin(t)+(p.split?halfSign*p.separation/130:0);
  return [x*Math.cos(a)-y*Math.sin(a),x*Math.sin(a)+y*Math.cos(a)];
 }
 function paths(raw={},n=480){
  const p=validate(raw);
  if(!Number.isInteger(n)||n<30||n>8000)throw Error('Invalid samples');
  if(!p.split)return [Array.from({length:n+1},(_,i)=>mappedPoint(i*TAU/n,p))];
  // Remove two positive-size open gaps centered on t=0 and pi.
  // These are two labeled arcs; their geometric images might overlap when bend >= 1.
  const gap=.10, arr=[];
  for(const [start,end,sign] of [[gap,Math.PI-gap,1],[Math.PI+gap,TAU-gap,-1]]){
    arr.push(Array.from({length:Math.floor(n/2)+1},(_,i)=>mappedPoint(start+(end-start)*i/Math.floor(n/2),p,sign)));
  }
  return arr;
 }
 function distance(a,b){return Math.hypot(a[0]-b[0],a[1]-b[1]);}
 function circleProperties(radius){if(!(radius>0))throw Error('Radius must be positive');return {radius,diameter:2*radius,perimeter:TAU*radius,area:Math.PI*radius*radius,ratio:Math.PI};}
 function classification(raw={}){
  const p=validate(raw),degenerate=p.collapse>=1-1e-12;
  const circle=!p.split&&p.bend===0&&!degenerate;
  const jordan=!p.split&&p.bend<1&&!degenerate;
  const injective=p.bend<1&&!degenerate;
  const pieces=p.split?2:1;
  const exactSimilarity=circle; // transformed whole shape is a positive-scale rotated circle
  const stage=degenerate?'collapsed':p.split?'split':p.bend>=1?'pinched':p.bend>0?'bent':'similar';
  const pointMapInfo=!injective?'Many source positions can map to one visible point':p.split?'Each surviving arc has a one-to-one parametrization':'The full boundary has a one-to-one parametrization';
  return {stage,circle,exactSimilarity,jordan,injective,abstractPieces:pieces,ledger:p.ledger,
   reconstructOriginal:p.ledger, // only because the ledger explicitly saves the original specification
   pointMapInfo,
   note:degenerate?'The image has collapsed. Geometry alone cannot reconstruct the circle. A separate source record still can.':p.split?'Two labeled arcs remain. The original circle connection is cut; at strong bends their visible traces may overlap.':p.bend>=1?'At bend 1, several parameter values reach the origin. Beyond it, further crossings occur; one-to-one recognition fails.':p.bend>0?'Circularity is lost but the boundary is still a simple closed loop.':'Uniform scale and rotation keep the boundary exactly circular.'};
 }
 function sampleCollisions(a){
  if(a<1)return 0;
  // r(t)=0 iff cos(3t)=-1/a; at a=1 there are three roots, beyond 1 six.
  return a===1?3:6;
 }
 function witness(raw={}){
  const p=validate(raw);return p.ledger?{source:'unit circle with ordered angles',recipe:{...p},note:'Source preserved externally; this does NOT make a lossy point map invertible.'}:null;
 }
 return Object.freeze({TAU,defaults,validate,radial,sourcePoint,mappedPoint,paths,distance,circleProperties,classification,sampleCollisions,witness});
});
