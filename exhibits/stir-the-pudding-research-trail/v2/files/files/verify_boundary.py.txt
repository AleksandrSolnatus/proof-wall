#!/usr/bin/env python3
"""Independent small-model oracle for PUDDING-15 outputs.

This checker intentionally does not import the generator/solver module. It
enumerates Boolean gate semantics directly for every saved small-chain case.
Run: python3 verify_boundary.py results.json
"""
import json
import sys
from itertools import product
from pathlib import Path


def check_small(rec):
    n=rec['n']
    flips=rec['flips']
    observed={}
    sat=0
    for bits in product((0,1), repeat=3*n+1):
        ports=bits[:n+1]
        priv=bits[n+1:]
        good=True
        for i in range(n):
            t,u=priv[2*i:2*i+2]
            if u != (ports[i] & t) or ports[i+1] != (ports[i] ^ flips[i]):
                good=False
                break
        if good:
            sat+=1
            key=str((ports[0],ports[-1]))
            observed[key]=observed.get(key,0)+1
    assert observed==rec['weighted_boundary'],(n,flips,observed)
    assert sat==rec['full_assignments']
    assert len(observed)==rec['distinct_endpoint_pairs']
    return 1


def main():
    source=Path(sys.argv[1] if len(sys.argv)>1 else 'results.json')
    obj=json.loads(source.read_text())
    assert obj['id']=='PUDDING-15-BOUNDARY-COLLAPSE-2026-11-08'
    checked=sum(check_small(x) for x in obj['chain_small_checks'])
    assert checked==42
    for row in obj['bundle_results']:
        k=row['wires']
        assert row['whole_macro_table_rows']==4**k
        assert row['factorized_local_table_rows']==4*k
        assert row['true_boundary_rows']==2**k
        assert row['satisfying_full_assignments']==4**k
        assert row['truth_table_to_factored_row_ratio']==(4**k)/(4*k)
    assert obj['scalar_counterexample']['sat_conjunction'] is False
    assert obj['scalar_counterexample']['scalar_and_prediction'] is True
    assert sum(obj['exhaustive_elimination_stage_checks'].values())==52
    print('PASS: 42 independently enumerated direct-gate small-chain cases; 8 boundary-table sizes; scalar counterexample and saved stage counts checked.')


if __name__=='__main__':main()
