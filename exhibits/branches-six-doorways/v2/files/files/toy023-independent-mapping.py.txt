"""Check a visual transcription from printed page 178; imports no packet code."""
import itertools, json, hashlib
from pathlib import Path
p = Path(__file__).resolve().parent
paper = [(1,2,3),(-1,-2,-3),(-1,2,4),(1,-2,-4),(1,-3,4),(-1,3,-4),(2,-3,-4),(-2,3,4)]
target = [(2,3,4),(1,-2,4),(-1,-2,3),(1,2,-3),(-1,-3,4),(1,3,-4),(-1,2,-4),(-2,-3,-4)]
mapping = {1:-1,2:-2,3:3,4:4}
image = [frozenset(mapping[abs(x)]*(1 if x>0 else -1) for x in c) for c in paper]
require = lambda ok: None if ok else (_ for _ in ()).throw(ValueError('failed check'))
require(set(image)==set(map(frozenset,target)))
covered = [[] for _ in target]
for a in itertools.product((False,True), repeat=4):
    failures = [i for i,c in enumerate(target) if not any(a[abs(x)-1]==(x>0) for x in c)]
    require(len(failures)==1)
    covered[failures[0]].append(a)
require(all(len(a)==2 for a in covered))
r={'paper_formula_visual_transcription':paper,'target':target,'signed_map':mapping,'paper_to_target_1_based':[list(map(frozenset,target)).index(c)+1 for c in image], 'map_exact':True, 'assignments_checked':16,'each_clause_unique_violation_witnesses':[len(a) for a in covered], 'all_eight_axioms_necessary':True, 'published_minimum_total_lines':19,'derived_minimum_inferences':11, 'minimum_search_rerun':False,'pdf_sha256':hashlib.sha256((p/'Peitl-Szeider-2023.pdf').read_bytes()).hexdigest()}
(p/'INDEPENDENT-MAPPING.json').write_text(json.dumps(r,indent=2)+'\n')
print(json.dumps(r,indent=2))
