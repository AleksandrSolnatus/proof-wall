"""Reviewer-authored semantic/certificate/input-binding checks; no source helper import.
Does not independently regenerate the exhaustive dynamic-programming tables.
"""
from pathlib import Path
from itertools import combinations
from collections import Counter
import hashlib, json, struct, time

ROOT = Path(__file__).resolve().parent
SOURCE = ROOT / 'source'

def require(value, reason):
    if not value:
        raise ValueError(reason)

def rejected_assignments(literals):
    result = 0
    for a in range(16):
        if not any(bool(a & (1 << (abs(l)-1))) == (l > 0) for l in literals):
            result |= 1 << a
    return result

def check_certificate(formula, certificate):
    lines = [frozenset(c) for c in formula]
    parents = {}
    uses = Counter()
    for left, right, pivot, conclusion in certificate:
        require(isinstance(left,int) and isinstance(right,int), 'integer parents')
        require(0 <= left < len(lines) and 0 <= right < len(lines), 'acyclic parents')
        require(pivot in lines[left] and -pivot in lines[right], 'opposite pivot')
        expected = (lines[left] - {pivot}) | (lines[right] - {-pivot})
        require(expected == frozenset(conclusion), 'exact resolvent')
        parents[len(lines)] = [left,right]
        uses.update([left,right])
        lines.append(frozenset(conclusion))
    require(lines[-1] == frozenset(), 'empty terminal')
    reached = set()
    stack = [len(lines)-1]
    while stack:
        current = stack.pop()
        if current in reached:
            continue
        reached.add(current)
        stack.extend(parents.get(current, []))
    require(reached == set(range(len(lines))), 'every source/derived node contributes')
    return {'inferences':len(certificate),'shared_original_input_ids':[i for i in range(len(formula)) if uses[i]>1],
            'shared_derived_nodes':[i for i in range(len(formula),len(lines)) if uses[i]>1],
            'extra_premise_uses':sum(v-1 for v in uses.values())}

def main():
    start = time.perf_counter()
    old = json.loads((SOURCE/'source_checkpoint.json').read_text())
    formula = old['formula']
    require(len(formula)==8 and all(len(c)==3 for c in formula), 'input shape')
    encoded = [sum(1 << (2*(abs(l)-1)+(l<0)) for l in c) for c in formula]
    require(encoded == list(map(int,(SOURCE/'input.txt').read_text().split())), 'C++/certificate input binding')
    excluded = [rejected_assignments(c) for c in formula]
    require(all(mask.bit_count()==2 for mask in excluded), 'each excludes two')
    require(sum(excluded)==65535 and (lambda masks: all(not (a&b) for a,b in combinations(masks,2)))(excluded), 'partition')
    universe = [1,-1,2,-2,3,-3,4,-4]
    all_clauses = [frozenset(universe[j] for j in range(8) if code & (1<<j)) for code in range(256)]
    falsities = [rejected_assignments(c) for c in all_clauses]
    isolated = []
    for subset in range(1,256):
        bad = 0
        for j in range(8):
            if subset & (1<<j): bad |= excluded[j]
        for clause,mask in zip(all_clauses,falsities):
            if mask==bad:
                isolated.append((subset,sorted(clause)))
                require(subset.bit_count()==1 or subset==255 and not clause, 'nontrivial isolated cut')
    require(len(isolated)==9, 'isolated count')
    mixed = []
    for input_id in range(8):
        domain = 65535 ^ excluded[input_id]
        other = [j for j in range(8) if j!=input_id]
        categories=Counter()
        for subset in range(128):
            bad = 0
            for j,original_id in enumerate(other):
                if subset & (1<<j):bad |= excluded[original_id]
            for clause,mask in zip(all_clauses,falsities):
                if (bad & domain)!=(mask & domain):continue
                if subset==0:categories['empty']+=1
                elif subset.bit_count()==1:
                    require(clause==frozenset(formula[other[subset.bit_length()-1]]),'mixed singleton identity')
                    categories['singleton']+=1
                elif subset==127:
                    require(not clause,'mixed full empty')
                    categories['full']+=1
                else:raise ValueError('nontrivial mixed cut')
        require(categories=={'empty':178,'singleton':7,'full':1},'mixed categories')
        mixed.append(dict(categories))
    graph = check_certificate(formula,old['certificate'])
    require(graph=={'inferences':11,'shared_original_input_ids':[0,2,3,6],'shared_derived_nodes':[],'extra_premise_uses':4}, 'certificate graph')
    rejected=0
    for mode in ['future_parent','wrong_resolvent']:
        bad=json.loads(json.dumps(old['certificate']))
        if mode=='future_parent':bad[0][0]=8
        else:bad[-1][3]=[1]
        try:check_certificate(formula,bad)
        except ValueError:rejected+=1
    require(rejected==2,'corrupt certificate accepted')
    expected={(i,j,u,v) for i,j in combinations(range(8),2) for u,v in [(2,4),(3,3),(4,2)]}
    actual={tuple(map(int,p.stem.split('_'))) for p in (SOURCE/'tables').glob('*.bin')}
    require(actual==expected,'84 table coverage')
    lower_states=0
    for i,j,U,V in sorted(expected):
        data=(SOURCE/'tables'/f'{i}_{j}_{U}_{V}.bin').read_bytes()
        require(len(data)==64*(U+1)*(V+1)*32,'table size')
        for u in range(2,U+1):
            for v in range(2,V+1):
                if u+v<=6:
                    offset=((63*(U+1)+u)*(V+1)+v)*32
                    require(not (int.from_bytes(data[offset:offset+32],'little') & 1), 'empty clause in saved table')
                    lower_states+=1
    out={'status':'PASS','formula':formula,'model':'one source identity per original input clause; duplicate uses count as sharing',
         'isolated_cut_matches':len(isolated),'mixed_cut_categories_by_input':mixed,'certificate_graph':graph,
         'corruptions_rejected':rejected,'exact_table_inventory':len(expected),'saved_final_state_reads':lower_states,
         'wall_seconds':time.perf_counter()-start,'source_checkpoint_sha256':hashlib.sha256((SOURCE/'source_checkpoint.json').read_bytes()).hexdigest(),
         'limits':'Reviewer-authored semantics, proof replay and saved-table coverage/read checks. Does not independently regenerate table recurrence or establish unrestricted minimum proof length.'}
    (ROOT/'independent-check.json').write_text(json.dumps(out,indent=2)+'\n')
    print(json.dumps(out,indent=2))

if __name__=='__main__':main()
