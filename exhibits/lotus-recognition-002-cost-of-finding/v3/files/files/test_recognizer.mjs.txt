import assert from 'node:assert/strict';
import {inverse,verifyInverse,repair} from './baseline/research/rising-tide-001/space/demo.mjs';
import {findEdit,meter,Learner,total,key} from './learner.mjs';
import {edit,makeSequence} from './fixtures.mjs';
let pairs=0,edges=0,budgetChecks=0,nonEdges=0;
for(let n=1;n<=3;n++) {
  const all=[];
  for(let z=0;z<2**(n*n);z++) {
    const A=Array.from({length:n},(_,i)=>Array.from({length:n},(_,j)=>(z>>(i*n+j))&1));
    const B=inverse(A);if(B)all.push({A,B});
  }
  for(const {A,B} of all)for(const {A:C} of all) {
    const possible=[];
    for(let i=0;i<n;i++)for(let j=0;j<n;j++)if(i!==j&&key(edit(A,i,j))===key(C))possible.push([i,j]);
    const m=meter(),r=findEdit(A,C,m);pairs++;
    assert.equal(r.kind==='edit',possible.length>0);
    assert.equal(total(m),r.used);
    if(r.kind==='edit') {
      assert(possible.some(([i,j])=>i===r.i&&j===r.j));
      assert(verifyInverse(C,repair(A,B,C,r.i,r.j)));edges++;
    } else nonEdges++;
    for(const budget of [0,1,3,Math.max(0,r.used-1),r.used]) {
      const bm=meter(),b=findEdit(A,C,bm,budget);budgetChecks++;
      assert(b.used<=budget);assert.equal(total(bm),b.used);
      if(b.kind==='edit')assert(possible.some(([i,j])=>i===b.i&&j===b.j));
      if(budget===r.used)assert.deepEqual(b,r);
    }
  }
}
// API isolation, caller mutation, FIFO capacity, accounting and exact hits.
const seq=makeSequence({n:8,seed:41,capacity:4,q:1,kind:'chain'});
const learner=new Learner('transfer_scan',4);
let prev=0;
for(const raw of seq.problems) {
  const p=structuredClone(raw),out=learner.solve(p.A,p.b,p.h,p.d);
  assert(out.total>=prev);prev=out.total;
  assert.equal(out.total,Object.values(out.ledger).reduce((a,b)=>a+b,0));
  assert(learner.archive.length<=4);assert(learner.cells>=0);
  for(const r of learner.archive)assert(verifyInverse(r.A,r.B));
  p.A[0][0]^=1;for(const r of learner.archive)assert(verifyInverse(r.A,r.B));
}
const p=seq.problems.at(-1),hit=learner.solve(p.A,p.b,p.h,p.d);
assert.equal(hit.route,'answer-hit');
assert(learner.stats.transfers>0);
const budgetLearner=new Learner('transfer_budget',4);
for(const p of seq.problems)budgetLearner.solve(p.A,p.b,p.h,p.d);
assert(budgetLearner.stats.searchUnits <= (4*64+2*64)*seq.problems.length);
console.log(JSON.stringify({passed:true,pairs,recognizedEdges:edges,nonEdges,budgetChecks,
  callerIsolation:true,retainedInverseChecks:true,exactHit:true},null,2));
