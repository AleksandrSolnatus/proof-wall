import original_math01 as m
P=pathlib.Path(__file__).resolve().parent
O=P.parent/'math02-intake/unpacked/MATH-01'
MODELS=m.MODELS[:6]
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def components(cells):
 parent=list(range(len(cells)))
 def root(i):
  while parent[i]!=i:i=parent[i]
  return i
 for i,a in enumerate(cells):
  for j in range(i):
   b=cells[j];x=a['signs'];y=b['signs'];short,long=sorted((x,y),key=len)
   if m.dist1(x,y) or any(short==long[k:k+len(short)] for k in range(len(long)-len(short)+1)):
    parent[root(i)]=root(j)
 groups=collections.defaultdict(list)
 for i in range(len(cells)):groups[root(i)].append(i)
 return sorted(groups.values(),key=lambda g:(-sum(len(cells[i]['signs']) for i in g),g))
def profile(cells):
 counts=collections.Counter(s for c in cells for s in c['signs']);n=sum(counts.values())
 return dict(cells=len(cells),tokens=n,types=len(counts),length_histogram=dict(sorted(collections.Counter(len(c['signs']) for c in cells).items())),top_symbol_fraction=max(counts.values())/n,collision_probability=sum(v*(v-1) for v in counts.values())/(n*(n-1)),duplicate_cell_excess=len(cells)-len({tuple(c['signs']) for c in cells}))
def score(cells,train,test):
 prec=m.prev_cells(cells);base,role=m.train(cells,train,prec)
 n=sum(len(cells[i]['signs']) for i in test)
 return {model:sum(-math.log2(m.tokenprob(model,cells,i,j,base,role,prec)) for i in test for j in range(len(cells[i]['signs'])))/n for model in MODELS}
def contrast(v):
 a=min(v[k] for k in MODELS if k[0] in 'ABC');b=min(v[k] for k in MODELS if k[0] in 'DE')
 return dict(math_advantage=b-a,positive_math_call=b-a>0.05,advantage_including_unigram=min(b,v['F_unigram'])-a,best_model=min(v,key=v.get))
def wilson(k,n):
 if not n:return None
 z=1.95996398454;p=k/n;den=1+z*z/n;c=(p+z*z/(2*n))/den;h=z*math.sqrt(p*(1-p)/n+z*z/(4*n*n))/den
 return [c-h,c+h]

  replay[var]={'max_abs_error':err,'scores':scores,'profile':profile(cells)}
 result={'status':'INCOMPLETE_CALIBRATION','original_manifest_files_verified':len(integrity),'original_replay':replay,'classes':{},'synthetic':{},'python':platform.python_version()}
 for cl in ['math','language','registry']:

  if pr['types']>45:cr['status']='INELIGIBLE_ALPHABET';continue

   test=[i for i,c in enumerate(cells) if c['source']==source];touch=set(test);purge={i for g in groups if touch.intersection(g) for i in g};train=[i for i in range(len(cells)) if i not in purge]
   ntrain=sum(len(cells[i]['signs']) for i in train);nsource=len({cells[i]['source'] for i in train});ntest=sum(len(cells[i]['signs']) for i in test)
