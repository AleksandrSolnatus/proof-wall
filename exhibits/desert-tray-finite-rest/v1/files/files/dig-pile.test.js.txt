const sum=m=>m.heights.reduce((a,b)=>a+b,0),energy=m=>m.heights.reduce((a,b)=>a+b*b,0);
function invariant(m){const q=m.metrics();assert.equal(sum(m)+m.scoop,m.total);assert.equal(q.aggregateUnits,m.total);assert.equal(q.trayUnits,sum(m));assert.equal(q.scoopUnits,m.scoop);assert.ok(m.scoop>=0&&Number.isInteger(m.scoop));assert.ok(m.heights.every(h=>h>=0&&Number.isInteger(h)));out.checks++;}
function stable(m){assert.equal(m.tool,null);for(let z=0;z<m.depth;z++)for(let x=0;x<m.width;x++)for(const [dx,dz,limit]of[[1,0,3],[0,1,3],[1,1,4],[-1,1,4]]){const nx=x+dx,nz=z+dz;if(nx<0||nx>=m.width||nz>=m.depth)continue;assert.ok(Math.abs(m.heights[z*m.width+x]-m.heights[nz*m.width+nx])<=limit);}}
function step(m){const before=energy(m),wasActive=m.tool&&m.tool.mode!=='push',moved=m.step();out.steps++;if(moved)out.movingSteps++;invariant(m);if(!wasActive){const after=energy(m);assert.ok(moved?after<before:after===before,'energy while no dig/pile');}return moved;}
function settle(m){m.release();let n=0;do{step(m);n++;assert.ok(n<10000,'rest within 10k steps');}while(!m.resting);stable(m);return n;}
function diskMass(m,x,z,r){let s=0;for(let k=0;k<m.heights.length;k++)if(Math.hypot(k%m.width+.5-x,Math.floor(k/m.width)+.5-z)<r)s+=m.heights[k];return s;}
