def truth(F):
 vs=sorted({abs(l) for c in F for l in c});out=[]
 for bits in itertools.product([False,True],repeat=len(vs)):
  a=dict(zip(vs,bits))
  if all(any(a[abs(l)]==(l>0) for l in c) for c in F):out.append(a)
 return out

def replay(F,result,kind):
 nodes=[frozenset(c) for c in F];assert len(nodes)==len(F)
 source=hashlib.sha256(json.dumps(F,separators=(',',':')).encode()).hexdigest();assert result['source_sha256']==source;prefix=source
 events=result['certificates'] if kind=='companion' else [p['certificate'] for p in result['proofs']]
 ms=truth(F)
 for step,c in enumerate(events):
  assert c['source_sha256']==source and c['prefix_sha256']==prefix and type(c['step']) is int and c['step']==step
  ids=c['parents'] if kind=='companion' else [c['left'],c['right']];assert len(ids)==2 and all(type(i) is int and 0<=i<len(nodes) for i in ids)
  x=c['pivot'];assert type(x) is int and x!=0
  a,b=[nodes[i] for i in ids];assert x in a and -x in b
  derived=(a-{x})|(b-{-x});assert not any(-l in derived for l in derived)
  assert all(type(l) is int and l!=0 for l in c['clause']);assert len(set(c['clause']))==len(c['clause']) and derived==frozenset(c['clause'])
  assert all(any(a[abs(l)]==(l>0) for l in derived) for a in ms)
  nodes.append(derived);payload=json.dumps(c,sort_keys=True,separators=(',',':'),ensure_ascii=False).encode();prefix=hashlib.sha256(prefix.encode()+b'\n'+payload).hexdigest()
 exported=[frozenset(n['clause']) for n in result['nodes']] if kind=='companion' else list(map(frozenset,result['nodes']))
 assert exported==nodes and result['prefix_sha256']==prefix
 if kind=='companion':assert result['original']==F;units=sorted(int(v) if b else -int(v) for v,b in result['facts'].items())
 else:units=result['units']
 assert all(frozenset([l]) in nodes for l in units)
 return {'models':len(ms),'proof_steps':len(events),'units':units}

def dp(F):
 nodes=[frozenset(c) for c in F];ids={c:i for i,c in enumerate(nodes)};active=set(range(len(F)));proof=[];attempts=0
 for v in sorted({abs(l) for c in F for l in c}):
  plus=sorted(i for i in active if v in nodes[i]);minus=sorted(i for i in active if -v in nodes[i]);nextactive={i for i in active if v not in nodes[i] and -v not in nodes[i]}
  for a,b in itertools.product(plus,minus):
   attempts+=1;c=(nodes[a]-{v})|(nodes[b]-{-v})
   if any(-l in c for l in c):continue
   if c not in ids:
    ids[c]=len(nodes);nodes.append(c);proof.append({'parents':[a,b],'pivot':v,'clause':sorted(c),'node_id':ids[c]})
   nextactive.add(ids[c])
   if not c:return proof,attempts
  active=nextactive
 raise AssertionError('No refutation found')

def check_dp(F,proof):
 nodes=list(map(frozenset,F))
 for c in proof:
  a,b=c['parents'];v=c['pivot'];assert 0<=a<len(nodes) and 0<=b<len(nodes);assert v in nodes[a] and -v in nodes[b]
  d=(nodes[a]-{v})|(nodes[b]-{-v});assert d==frozenset(c['clause']) and not any(-l in d for l in d);assert c['node_id']==len(nodes);nodes.append(d)
 assert not nodes[-1]

def parity_extract(F):
 groups={}
 for i,c in enumerate(F):
  vs=tuple(sorted({abs(l) for l in c}))
  if len(vs)!=3 or len(c)!=3:return None
  groups.setdefault(vs,[]).append((i,c))
 eq=[]
 for vs,rows in groups.items():
  if len(rows)!=4:return None
  allowed=[]
  for bits in itertools.product([0,1],repeat=3):
   a=dict(zip(vs,bits))
   if all(any(a[abs(l)]==(l>0) for l in c) for _,c in rows):allowed.append(bits)
  if len(allowed)!=4 or len({sum(b)%2 for b in allowed})!=1:return None
  eq.append({'variables':vs,'rhs':sum(allowed[0])%2,'original_clause_ids':[i for i,c in rows]})
 return eq

def main():
