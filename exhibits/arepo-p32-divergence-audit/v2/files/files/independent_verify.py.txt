def digest(b):return hashlib.sha256(b).hexdigest()
def same(a,b):
 if isinstance(a,(float,int)) and not isinstance(a,bool):assert math.isclose(a,b,rel_tol=2e-12,abs_tol=1e-10),(a,b)

def finite(x):

def actden(k,p,s):
 a=s['intervalActivity'];n=s['count'];ns=n*(.25/p['dt']);passes=ns*p['iterations'];spacing=p['spacing'];cap=p['speedCap']
 if k in STAGES:return .5*s['representedMass']*cap**2
 mapping={'speedCapPredictEvents':ns,'speedCapPostEvents':ns,'floorContactParticleSteps':ns,'reconstructionSamples':ns,'floorClampEvents':passes,'wallClampEvents':passes,'correctionCapEvents':a['repulsionEvaluations'],'attractionCapEvents':a['positiveAttractionEvaluations'],'reconstructionDeltaVSum':a['reconstructionSamples']*cap,'maxUncappedPairRequest':spacing,'maxAttractionRawRequest':spacing,'reconstructionDeltaVRms':cap,'reconstructionDeltaVMax':cap}
 if k in mapping:return mapping[k]
 if k in ['relationBranchTransitions','repulsionToShellTransitions','shellToRepulsionTransitions']:return a['repulsionEvaluations']+a['shellEvaluations']
 if k in ['pairCorrections','repulsionEvaluations','cohesionPairs','shellEvaluations','positiveAttractionEvaluations']:return passes
 if k in ['repulsionDemand','repulsionApplied']:return a['repulsionEvaluations']*spacing
 if k in ['attractionRawRequest','cohesionApplied']:return a['positiveAttractionEvaluations']*spacing

 return raw,raw/den if raw is not None and den else None
