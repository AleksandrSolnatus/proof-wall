 def check(self):

  assert self.supplied==sum(self.w)+self.drained+self.rejected
  assert all(a>=b for a,b in zip((self.supplied,self.drained,self.rejected),self.prev))
  self.prev=(self.supplied,self.drained,self.rejected)

 def flow(self,a,b,t):
  if t: key=str(a)+':'+str(b);self.flows[key]=self.flows.get(key,0)+t
 def step(self):
  self.tick+=1;n=self.n;w=self.w
  if self.rain and self.tick<=160:
   amount=n*n*K//80; q,r=divmod(amount,n)
   for i in range(n):
    a=n*(n-1)+i;offer=q+(i<r);t=0 if self.solid[a] else min(offer,K-w[a]);w[a]+=t;self.supplied+=offer;self.rejected+=offer-t;self.flow('rain',a,t);self.flow('rain','rejected',offer-t)
  self.check()
  for a in self.drains:
   t=min(w[a],Q);w[a]-=t;self.drained+=t;self.flow(a,'drain',t)
   if t and self.first is None:self.first=self.tick
  self.check()
  for a,b in self.down:
   t=min(w[a],K-w[b],Q);w[a]-=t;w[b]+=t;self.flow(a,b,t)
  self.check()
  for a,b in self.lateral:
   d=w[a]-w[b]
   if d<0:a,b=b,a;d=-d
   t=min(d//2,Q);w[a]-=t;w[b]+=t;self.flow(a,b,t)
  self.check();self.check()
 def row(self):return [self.tick,self.supplied,sum(self.w),self.drained,self.rejected,max(self.w),abs(self.supplied-sum(self.w)-self.drained-self.rejected)]
